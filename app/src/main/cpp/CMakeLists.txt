cmake_minimum_required(VERSION 3.22.1)
project(shufang_books C CXX)
set(CMAKE_C_STANDARD 99)
set(MOBI_SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/libmobi/src)
set(MOBI_FILES buffer compression debug index memory meta parse_rawml read structure util write opf xmlwriter)
set(MOBI_SOURCES)
foreach(name IN LISTS MOBI_FILES)
  list(APPEND MOBI_SOURCES ${MOBI_SOURCE}/${name}.c)
endforeach()
# Keep the LGPL library independently replaceable/relinkable, with its source in this repository.
add_library(mobi SHARED ${MOBI_SOURCES})
target_include_directories(mobi PUBLIC ${MOBI_SOURCE})
target_compile_definitions(mobi PRIVATE PACKAGE_VERSION="0.12" USE_XMLWRITER HAVE_UNISTD_H HAVE_STRDUP MOBI_INLINE=inline)
target_link_libraries(mobi PRIVATE z)
add_library(shufang_mobi SHARED mobi_bridge.c)
target_link_libraries(shufang_mobi PRIVATE mobi)
target_compile_options(shufang_mobi PRIVATE -Wall -Wextra -Werror)
if(ANDROID)
  target_link_options(mobi PRIVATE "-Wl,-z,max-page-size=16384")
  target_link_options(shufang_mobi PRIVATE "-Wl,-z,max-page-size=16384")
else()
  find_package(JNI REQUIRED)
  target_include_directories(shufang_mobi PRIVATE ${JNI_INCLUDE_DIRS})
endif()
# Rebuild/relink with a modified libmobi by replacing MOBI_SOURCE, then run :app:assembleDebug.
if(ANDROID)
  set(SPEECH_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../../../.build-cache/speech/runtime-1.12.21/jniLibs/${ANDROID_ABI})
  add_library(sherpa_speech SHARED IMPORTED)
  set_target_properties(sherpa_speech PROPERTIES IMPORTED_LOCATION ${SPEECH_ROOT}/libsherpa-onnx-c-api.so)
  add_library(speech_ort SHARED IMPORTED)
  set_target_properties(speech_ort PROPERTIES IMPORTED_LOCATION ${SPEECH_ROOT}/libonnxruntime.so)
  add_library(shufang_speech SHARED speech_bridge.cpp)
  target_compile_features(shufang_speech PRIVATE cxx_std_17)
  target_include_directories(shufang_speech PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/sherpa-onnx)
  target_link_libraries(shufang_speech PRIVATE sherpa_speech speech_ort)
  target_compile_options(shufang_speech PRIVATE -Wall -Wextra -Werror)
  target_link_options(shufang_speech PRIVATE "-Wl,-z,max-page-size=16384")
endif()
